(ns code.migrate-bang-plan
  (:require [code.migrate.bang :as bang]
            [tool.project :as project]))

(let [current-project (project/read-project "project.edn")
      units (bang/project-units current-project)
      migration (bang/plan units)]
  (pr-str
   {:valid (:valid migration)
    :records (:records migration)
    :files (:files migration)
    :changed/files (:changed/files migration)
    :first-records (vec (take 5 (:ledger migration)))
    :diagnostics (:diagnostics migration)}))
